
#include <LiquidCrystal.h>

// LCD CONNECTIONS: RS, E, D4, D5, D6, D7
LiquidCrystal lcd(7, 8, 5, 4, 3, 2);

// LED AND BUZZER PINS
const int greenLED = 10;
const int yellowLED = 11;
const int redLED = 12;
const int buzzer = 13;

// SENSOR INPUT PINS
const int currentSensor = A0;
const int vibrationSensor = A1;
const int temperatureSensor = A2;

// FAULT THRESHOLDS
const float TEMP_WARNING = 60.0;
const float TEMP_FAULT = 75.0;

const float CURRENT_WARNING = 6.0;
const float CURRENT_FAULT = 8.0;

const float VIBRATION_WARNING = 6.0;
const float VIBRATION_FAULT = 8.0;

void setup() {

  pinMode(greenLED, OUTPUT);
  pinMode(yellowLED, OUTPUT);
  pinMode(redLED, OUTPUT);
  pinMode(buzzer, OUTPUT);

  lcd.begin(16, 2);

  Serial.begin(9600);

  lcd.setCursor(0, 0);
  lcd.print("SMART EQUIPMENT");

  lcd.setCursor(0, 1);
  lcd.print("DIAGNOSTIC SYS");

  delay(2000);
  lcd.clear();
}

void loop() {

  // READ TEMPERATURE SENSOR (TMP36)
  int tempReading = analogRead(temperatureSensor);

  float voltage = tempReading * (5.0 / 1023.0);

  float temperatureC = (voltage - 0.5) * 100.0;

  // READ CURRENT POTENTIOMETER
  int currentReading = analogRead(currentSensor);

  float currentA = (currentReading / 1023.0) * 10.0;

  // READ VIBRATION POTENTIOMETER
  int vibrationReading = analogRead(vibrationSensor);

  float vibration = (vibrationReading / 1023.0) * 10.0;

  // CHECK FAULT CONDITIONS
  bool criticalFault =
    temperatureC >= TEMP_FAULT ||
    currentA >= CURRENT_FAULT ||
    vibration >= VIBRATION_FAULT;

  bool warningCondition =
    temperatureC >= TEMP_WARNING ||
    currentA >= CURRENT_WARNING ||
    vibration >= VIBRATION_WARNING;

  // DISPLAY AND ALERT LOGIC
  String status;

  if (criticalFault) {

    status = "FAULT";

    digitalWrite(greenLED, LOW);
    digitalWrite(yellowLED, LOW);
    digitalWrite(redLED, HIGH);

    tone(buzzer, 1000);

  }

  else if (warningCondition) {

    status = "WARNING";

    digitalWrite(greenLED, LOW);
    digitalWrite(yellowLED, HIGH);
    digitalWrite(redLED, LOW);

    noTone(buzzer);

  }

  else {

    status = "NORMAL";

    digitalWrite(greenLED, HIGH);
    digitalWrite(yellowLED, LOW);
    digitalWrite(redLED, LOW);

    noTone(buzzer);
  }

  // LCD FIRST LINE
  String line1 = "T:";
  line1 += String(temperatureC, 0);
  line1 += "C I:";
  line1 += String(currentA, 1);
  line1 += "A";

  lcd.setCursor(0, 0);
  lcd.print(line1);
  for (int i = line1.length(); i < 16; i++) {
    lcd.print(" ");
  }

  // LCD SECOND LINE
  String line2 = "V:";
  line2 += String(vibration, 1);
  line2 += " ";
  line2 += status;

  lcd.setCursor(0, 1);
  lcd.print(line2);
  for (int i = line2.length(); i < 16; i++) {
    lcd.print(" ");
  }

  // SERIAL MONITOR OUTPUT
  Serial.print("Temperature: ");
  Serial.print(temperatureC);
  Serial.print(" C | Current: ");
  Serial.print(currentA);
  Serial.print(" A | Vibration: ");
  Serial.print(vibration);
  Serial.print(" | Status: ");
  Serial.println(status);

  delay(500);
}
